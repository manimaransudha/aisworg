## ViewModel Architecture
### Purpose
To enforce strict view variable contracts, eliminate missing-variable errors, and standardize rendering.

### How/what to code

- The viewModels folder holds all the view models. 
- Each view has a corresponding ViewModel with:
  - **required keys**
  - **optional keys**
  - runtime validation logic

- If the view(ejs file) is under views/student/nav/dashboard, then the corresponding file in the viewModels folder is named as student_nav_dashboard.js

- Inside the view model file, a constant with <filenameVM> will be defined and exported. The required and optional keys will be inside the constant.  So, the code inside student_nav_dashboard.js will be 

``` code for student_nav_dashboard.js
import { createViewModel } from "../utils/viewModel.js";
export const student_nav_dashboardVM = createViewModel({
    required: ["title", "var1", "var2"],
    optional: ["flash"]
});
```
- Inside viewModels, there is a viewRegistry.js where the viewModels are registered as shown 

``` code to register a view model
import { student_nav_dashboardVM } from "./student_nav_dashboard.js";
export const viewModels = {
    "student/nav/dashboard": student_nav_dashboardVM,
}
```

## Controller Architecture

- Api controllers have to be under the routes/.../api/ folder or routes/.../web folder. There can be subfolders depending on functionality, like routes/student/api  etc. All controllers inside api/ will return a jsonn.  All controllers inside web/ will render a view. Both will will call the core/ for all business logic. This is to ensure there is no duplication of logic and make change management easy. 

- Web Controllers receive an initialized VM through middleware attachVM.

```code - web controller template
import {attachVM} from "middleware/attachVM.js";
import {renderView} from "utils/viewModel.js";
import {redirects} from 'middleware/redirects.js';

/** POST /admin/login — authenticate admin */
router.post('/login', attachVM("admin/login"), async (req, res) => {

//No dbcalls; All dbcalls have to go through dblayer
//Call core/ business logic to process all required logic and use the return values/json to generate the information needed for view
//Build the view model variables 
vm.req.title = "Some title"
vm.req.flash ="Some message" and so on. 

//Finally render the view as 
return renderView(req, res, "admin/home", req.vm); //the view to be rendered. 

//In case there is an error and a redirection has to happen, use the redirects middleware. 
if (!req.session?.admin) {
        return redirects.admin.home.authfail(req, res, "Invalid credentials");
    }
```
- Rendering is performed exclusively through `renderView` imported from viewModel.js

- Controllers needing roles will define it through the requireRole middleware 

```code
router.get('/', requireRole(['super', 'author']), attachVM("admin/subjects"), 
```

// dblayer/subjectsMaster.js
import { supabaseAdmin } from '../utils/supabaseAdmin.js';
import { supabaseApp } from '../utils/supabaseApp.js';
import {logger} from '../utils/logger.js';

function log(type, operation, message, err = null) {
    const prefix = 'subjects_master-' + operation + ': ' + message;
    if (type === 'error') logger.error(prefix, err);
    else logger.info(prefix);
}

function getClient(admin) {
    return admin ? supabaseAdmin : supabaseApp;
}

export const subjectsMasterDB = {

    /**
 * Fetch multiple records.
 *
 * Example:
 *   const result = await studentTestsDB.findAll({
 *       is_active: true,
 *       orderBy: { column: "created_at", ascending: false }
 *   });
 *
 *   if (result.error) {
 *       logger.error("Fetch failed:", result.error);
 *       return;
 *   }
 *
 *   const rows = result.data;   // ARRAY OF OBJECTS
 *   logger.debug("Rows:", rows);
 *
 * Returns:
 *   { data: [...], error: null }
 */

    async findAll({ admin = false, is_active = true, orderBy = null } = {}) {
        const db = getClient(admin);
        try {
            let query = db.from('subjects_master').select('*');

            if (true && is_active === true) {
                query = query.eq('is_active', true);
            }

            if (orderBy?.column) {
                query = query.order(orderBy.column, { ascending: orderBy.ascending !== false });
            } else {
                query = query.order('id', { ascending: true });
            }

            const { data, error } = await query;

            if (error) {
                log('error', 'findAll', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'findAll', err.message, err);
            return { error: err };
        }
    },

    /**
 * Fetch a single record by ID.
 *
 * Example:
 *   const result = await studentTestsDB.findOne("304cdc28-b073-47a1-8dc0-35b3");
 *
 *   if (result.error) {
 *       logger.error("Not found:", result.error);
 *       return;
 *   }
 *
 *   const row = result.data;   // SINGLE OBJECT
 *   logger.debug("Loaded row:", row);
 *
 * Returns:
 *   { data: {...}, error: null }
 */

    async findOne(id, { admin = false, is_active = true } = {}) {
        const db = getClient(admin);
        try {
            let query = db.from('subjects_master').select('*').eq('id', id);

            if (true && is_active === true) {
                query = query.eq('is_active', true);
            }

            const { data, error } = await query.single();

            if (error) {
                log('error', 'findOne', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'findOne', err.message, err);
            return { error: err };
        }
    },

    /**
 * Fetch records by matching fields.
 *
 * Example:
 *   const result = await studentTestsDB.findByParameters(
 *       { student_email: "x@y.com", assessment_id: "abc123" },
 *       { is_active: true }
 *   );
 *
 *   if (result.error) {
 *       logger.error("Query failed:", result.error);
 *       return;
 *   }
 *
 *   const rows = result.data;   // ARRAY OF OBJECTS
 *   logger.debug("Matched rows:", rows);
 *
 * Returns:
 *   { data: [...], error: null }
 */

    async findByParameters(params = {}, { admin = false, is_active = true, orderBy = null } = {}) {
        const db = getClient(admin);
        try {
            let query = db.from('subjects_master').select('*');

            Object.entries(params).forEach(([k, v]) => {
                query = query.eq(k, v);
            });

            if (true && is_active === true) {
                query = query.eq('is_active', true);
            }

            if (orderBy?.column) {
                query = query.order(orderBy.column, { ascending: orderBy.ascending !== false });
            }

            const { data, error } = await query;

            if (error) {
                log('error', 'findByParameters', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'findByParameters', err.message, err);
            return { error: err };
        }
    },

    /**
 * Insert a new row.
 *
 * Example:
 *   const result = await studentTestsDB.insert({
 *       student_email: "x@y.com",
 *       assessment_id: "A123",
 *       score: 0
 *   });
 *
 *   if (result.error) {
 *       logger.error("Insert failed:", result.error);
 *       return;
 *   }
 *
 *   const inserted = result.data;   // ARRAY containing 1 object
 *   logger.debug("Inserted row:", inserted);
 *
 * Returns:
 *   { data: [...], error: null }
 */

    async insert(payload, { admin = false } = {}) {
        const db = getClient(admin);
        try {
            const { data, error } = await db
                .from('subjects_master')
                .insert(payload)
                .select();

            if (error) {
                log('error', 'insert', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'insert', err.message, err);
            return { error: err };
        }
    },

   /**
 * Update rows by ID or predicate object.
 *
 * Example 1 — update by ID:
 *   const result = await studentTestsDB.update(
 *       "304cdc28-b073-47a1-8dc0-35b3",
 *       { score: 10 }
 *   );
 *
 * Example 2 — update by filter:
 *   const result = await studentTestsDB.update(
 *       { student_email: "x@y.com", assessment_id: "A123" },
 *       { score: 10 }
 *   );
 *
 * if (result.error) {
 *     logger.error("Update failed:", result.error);
 *     return;
 * }
 *
 * const updatedRows = result.data;   // ARRAY OF UPDATED OBJECTS
 * logger.debug("Updated:", updatedRows);
 *
 * Returns:
 *   { data: [...], error: null }
 */

    async update(where, payload, { admin = false } = {}) {
        const db = getClient(admin);
        try {
            let query = db.from('subjects_master').update(payload);

            if (typeof where === 'string' || typeof where === 'number') {
                query = query.eq('id', where);
            } else if (typeof where === 'object') {
                Object.entries(where).forEach(([k, v]) => {
                    query = query.eq(k, v);
                });
            }

            const { data, error } = await query.select();

            if (error) {
                log('error', 'update', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'update', err.message, err);
            return { error: err };
        }
    },

    /**
 * Soft-delete rows (sets is_active = false).
 *
 * Example:
 *   const result = await studentTestsDB.softDelete("304cdc28-b073-47a1-8dc0-35b3");
 *
 *   if (result.error) {
 *       logger.error("Soft delete failed:", result.error);
 *       return;
 *   }
 *
 *   const rows = result.data;   // ARRAY OF UPDATED ROWS
 *   logger.debug("Soft-deleted rows:", rows);
 *
 * Returns:
 *   { data: [...], error: null }
 */

    async softDelete(where, { admin = false } = {}) {
        if (!true) {
            logger.info("subjects_master: softDelete skipped — table has no is_active column");
            return { data: null };
        }

        const db = getClient(admin);

        try {
            let query = db.from('subjects_master').update({ is_active: false });

            if (typeof where === 'string' || typeof where === 'number') {
                query = query.eq('id', where);
            } else if (typeof where === 'object') {
                Object.entries(where).forEach(([k, v]) => (query = query.eq(k, v)));
            }

            const { data, error } = await query.select();

            if (error) {
                log('error', 'softDelete', error.message, error);
                return { error };
            }

            return { data };

        } catch (err) {
            log('error', 'softDelete', err.message, err);
            return { error: err };
        }
    },
